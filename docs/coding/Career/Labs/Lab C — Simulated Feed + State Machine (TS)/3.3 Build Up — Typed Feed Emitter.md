# 3.3 Build Up — Typed Feed Emitter

> **Now turn the EventEmitter into a real feed.** EventEmitter + OHLC type + Map pricing + interval.

## The Code

```typescript
// 3.3-feed-emitter.ts
import { EventEmitter } from "events"

interface OHLC {
  symbol: string
  open: number
  high: number
  low: number
  close: number
  timestamp: number
}

interface SignalData {
  symbol: string
  direction: "long" | "short"
}

class SimulatedFeed extends EventEmitter {
  private prices: Map<string, number> = new Map()
  private ticker: NodeJS.Timeout | null = null
  private tickCount = 0

  constructor(private symbols: string[]) {
    super()
  }

  start(): void {
    if (this.ticker) return  // already running

    for (const s of this.symbols) {
      this.prices.set(s, 100)  // seed price
    }

    this.ticker = setInterval(() => {
      this.tickCount++

      for (const symbol of this.symbols) {
        const current = this.prices.get(symbol) ?? 100
        const change = (Math.random() - 0.5) * 2
        const next = Math.max(1, current + change)
        this.prices.set(symbol, next)

        // Emit typed tick event
        this.emit("tick", {
          symbol,
          open: current,
          high: Math.max(current, next),
          low: Math.min(current, next),
          close: next,
          timestamp: Date.now(),
        } as OHLC)

        // Emit signal every 5 ticks randomly
        if (this.tickCount % 5 === 0 && Math.random() > 0.3) {
          this.emit("signal", {
            symbol,
            direction: Math.random() > 0.5 ? "long" : "short",
          } as SignalData)
        }
      }
    }, 500)

    console.log(`[feed] started: ${this.symbols.length} symbols`)
  }

  pause(): void {
    if (this.ticker) {
      clearInterval(this.ticker)
      this.ticker = null
      this.emit("pause")
      console.log("[feed] paused")
    }
  }

  resume(): void {
    if (!this.ticker) {
      this.start()
      this.emit("resume")
      console.log("[feed] resumed")
    }
  }

  stop(): void {
    if (this.ticker) clearInterval(this.ticker)
    this.ticker = null
  }
}

// === Demo ===
const feed = new SimulatedFeed(["AAPL", "MSFT"])

feed.on("tick", (tick: OHLC) => {
  console.log(`[tick] ${tick.symbol}: $${tick.close.toFixed(2)}`)
})

feed.on("signal", (signal: SignalData) => {
  console.log(`[signal] ${signal.symbol}: ${signal.direction}`)
})

feed.start()

setTimeout(() => {
  console.log("--- pausing ---")
  feed.pause()
  setTimeout(() => {
    console.log("--- resuming ---")
    feed.resume()
    setTimeout(() => {
      feed.stop()
      process.exit(0)
    }, 3000)
  }, 2000)
}, 4000)
```

## Run It

```bash
npx tsx 3.3-feed-emitter.ts
```

You'll see tick logs every 500ms, occasional signal logs, then pause/resume cycle.

## What It Is NOT

- **NOT real streaming.** Uses `setInterval` — real streamer uses WebSocket messages
- **NOT production logging.** Just `console.log` — real streamer uses structured logging

## Key Takeaway

The EventEmitter pattern means your feed (producer) and PositionMachine (consumer) **never import each other**. Feed doesn't know what's listening. Machines just `.on("tick", ...)`. This loose coupling is why you can add 10 new machines without touching the feed.
