# 4.2 Minimal Code — Route Ticks to Machines

> **Yaar, yeh file pe sab kuch milega — feed + routing + machine.** 
> Tick aata hai → Map lookup → machine handler. End-to-end demo.

## The Code

```typescript
// 4.2-route-ticks.ts
import { EventEmitter } from "events"

interface OHLC {
  symbol: string
  open: number
  high: number
  low: number
  close: number
  timestamp: number
}

type Status = "idle" | "watching" | "in_position" | "exited"

class PositionMachine {
  private status: Status = "idle"

  constructor(public symbol: string) {}

  getStatus(): Status { return this.status }

  watch(): void {
    if (this.status === "idle") this.status = "watching"
    console.log(`  ${this.symbol}: → watching`)
  }

  onTick(price: number): void {
    console.log(`  ${this.symbol}[tick] $${price.toFixed(2)} (${this.status})`)
  }
}

class SimulatedFeed extends EventEmitter {
  private tickCount = 0
  private ticker: NodeJS.Timeout | null = null

  constructor(private symbols: string[]) {
    super()
  }

  start(): void {
    if (this.ticker) return
    this.ticker = setInterval(() => {
      this.tickCount++
      for (const symbol of this.symbols) {
        const price = 100 + Math.random() * 50

        this.emit("tick", {
          symbol,
          open: price - 1,
          high: price + 1,
          low: price - 1,
          close: price,
          timestamp: Date.now(),
        } as OHLC)
      }
    }, 1000)

    console.log(`[feed] started: ${this.symbols.join(", ")}`)
  }

  stop(): void {
    if (this.ticker) clearInterval(this.ticker)
    this.ticker = null
  }
}

// === MAIN: WIRE EVERYTHING ===

const feed = new SimulatedFeed(["AAPL", "MSFT", "GOOGL"])

// Create machine per symbol
const managers = new Map<string, PositionMachine>([
  ["AAPL", new PositionMachine("AAPL")],
  ["MSFT", new PositionMachine("MSFT")],
  ["GOOGL", new PositionMachine("GOOGL")],
])

// Subscribe each machine to its watch state
for (const mgr of managers.values()) {
  mgr.watch()
}

// === THE ROUTING: feed emits → Map lookup → machine ===
feed.on("tick", (tick: OHLC) => {
  const handler = managers.get(tick.symbol)   // O(1) lookup
  if (handler) {
    handler.onTick(tick.close)  // route to correct machine
  } else {
    console.log(`[warn] no machine for ${tick.symbol}`)
  }
})

feed.start()

setTimeout(() => {
  console.log("\n=== All states ===")
  for (const [symbol, mgr] of managers) {
    console.log(`${symbol}: ${mgr.getStatus()}`)
  }
  feed.stop()
  process.exit(0)
}, 5000)
```
