# 5.2 Build Up — Full Pause Resume Demo

> **Ab poora pause/resume demo banao with state preservation.**

## Full Demo Script

```typescript
// 5.2-pause-resume-demo.ts
import { EventEmitter } from "events"

// === Types ===
interface OHLC {
  symbol: string
  close: number
  timestamp: number
}

type Status = "idle" | "watching" | "in_position" | "exited"

// === PositionMachine ===
class PositionMachine {
  private status: Status = "idle"
  private entryPrice: number | null = null

  constructor(public symbol: string) {}

  getStatus(): Status {
    return this.status
  }

  onTick(price: number): void {
    console.log(`  [tick] ${this.symbol}: $${price.toFixed(2)} (${this.status})`)
  }

  onSignal(direction: "long" | "short"): void {
    console.log(`  [signal] ${this.symbol}: ${direction} (${this.status})`)

    if (this.status === "watching") {
      this.status = "in_position"
      this.entryPrice = 150  // simplified — in reality uses tick price
      console.log(`  ${this.symbol}: watching → in_position (entry: $${this.entryPrice})`)
    } else if (this.status === "in_position") {
      this.status = "exited"
      console.log(`  ${this.symbol}: in_position → exited (entry was $${this.entryPrice})`)
      this.entryPrice = null
    } else {
      console.log(`  ${this.symbol}: signal ignored (not watching/in_position)`)
    }
  }

  watch(): void {
    if (this.status === "idle") {
      this.status = "watching"
      console.log(`  ${this.symbol}: idle → watching`)
    }
  }

  reset(): void {
    this.status = "idle"
    this.entryPrice = null
    console.log(`  ${this.symbol}: reset to idle`)
  }
}

// === SimulatedFeed with EventEmitter ===
const ALLOWED: Record<Status, Status[]> = {
  idle: ["watching"],
  watching: ["in_position", "idle"],
  in_position: ["exited", "idle"],
  exited: ["idle"],
}

class SimulatedFeed extends EventEmitter {
  private prices: Map<string, number> = new Map()
  private ticker: NodeJS.Timeout | null = null
  private tickCount = 0

  constructor(private symbols: string[]) {
    super()
  }

  start(): void {
    if (this.ticker) return
    for (const s of this.symbols) this.prices.set(s, 100)

    this.ticker = setInterval(() => {
      this.tickCount++
      for (const symbol of this.symbols) {
        const current = this.prices.get(symbol) ?? 100
        const next = current + (Math.random() - 0.5) * 2
        this.prices.set(symbol, next)

        this.emit("tick", { symbol, close: next, timestamp: Date.now() })

        // Signal once every 5 ticks, only for AAPL
        if (this.tickCount % 5 === 0 && symbol === "AAPL") {
          this.emit("signal", { symbol, direction: "long" })
        }
      }
    }, 500)
  }

  pause(): void {
    if (this.ticker) {
      clearInterval(this.ticker)
      this.ticker = null
      this.emit("pause")
      console.log("[feed] paused (disconnect)")
    }
  }

  resume(): void {
    if (!this.ticker) {
      this.start()
      this.emit("resume")
      console.log("[feed] resumed (reconnect)")
    }
  }

  stop(): void {
    if (this.ticker) clearInterval(this.ticker)
    this.ticker = null
  }
}

// === Main ===
const SYMBOLS = ["AAPL", "MSFT", "GOOGL"]
const feed = new SimulatedFeed(SYMBOLS)
const managers = new Map(
  SYMBOLS.map(s => [s, new PositionMachine(s)])
)

// Wire routing
feed.on("tick", (tick: OHLC) => {
  managers.get(tick.symbol)?.onTick(tick.close)
})

feed.on("signal", (sig: { symbol: string; direction: "long" | "short" }) => {
  managers.get(sig.symbol)?.onSignal(sig.direction)
})

feed.on("pause", () => {
  console.log(">>> All machines: holding state during disconnect")
})

feed.on("resume", () => {
  console.log(">>> All machines: resuming tick reception")
})

// Start
console.log("=== Starting Simulation ===\n")
for (const mgr of managers.values()) mgr.watch()
feed.start()

// Pause after 3s (AAPL should enter position before pause)
setTimeout(() => {
  console.log("\n--- PAUSE: Simulating disconnect ---")
  feed.pause()
  console.log(`AAPL state at pause: ${managers.get("AAPL")?.getStatus()}`)
}, 3000)

// Resume after 3s more
setTimeout(() => {
  console.log("\n--- RESUME: Simulating reconnect ---")
  feed.resume()
}, 6000)

// End
setTimeout(() => {
  console.log("\n=== FINAL STATES ===")
  for (const mgr of managers.values()) {
    console.log(`${mgr.symbol}: ${mgr.getStatus()}`)
  }
  console.log("\nState preserved across disconnect/resconnect ✓")
  feed.stop()
  process.exit(0)
}, 9000)
```

**Key thing to watch for:**
```
--- PAUSE: Simulating disconnect ---
[feed] paused (disconnect)
>>> All machines: holding state during disconnect
AAPL state at pause: in_position     ← ← ← STATE SURVIVED

--- RESUME: Simulating reconnect ---
[feed] resumed (reconnect)
>>> All machines: resuming tick reception
  [tick] AAPL: $100.43 (in_position)  ← ← ← STILL in_position after resume
```

## Done When

- [ ] Feed pauses — no ticks reach machines
- [ ] Machine states persist through pause
- [ ] Feed resumes — ticks flow again
- [ ] Machine state unchanged by disconnect

## Common Mistake

```typescript
// ❌ State reset on reconnect
resume() {
  this.machine.reset()  // NO — this loses position!
  this.start()
}
```

## Bridge

This **exact pause/resume pattern** is what TradeStack's `streamer.ts` needs to survive:
- Network blip → WebSocket `onclose` → machines keep state
- Reconnect-with-backoff → new WebSocket → machines still alive
- No state loss — critical for paper trading where every tick matters
