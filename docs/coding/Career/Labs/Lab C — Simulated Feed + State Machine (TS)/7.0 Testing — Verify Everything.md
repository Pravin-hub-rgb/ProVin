# 7.0 Testing — Verify Everything

> **Code may run, but if you don't verify, nothing really happened.** Write test cases — happy path + edge cases + invalid paths.

## Test File

```typescript
// 7.0-test.ts
/**
 * Lab C — Test Suite
 */

import { EventEmitter } from "events"
// Import from the final assembly (or just copy the classes)

// === Test: Double Entry Prevention ===
function testDoubleEntryPrevention() {
  console.log("\n=== Test: Double Entry Prevention ===")

  const mgr = new PositionMachine("TEST")
  mgr.watch()                    // idle → watching ✓
  const firstEnter = mgr.enter(150)     // watching → in_position ✓
  const secondEnter = mgr.enter(155)    // ❌ INVALID (in_position → in_position not allowed)

  console.log("First enter succeeded:", firstEnter ? "PASS" : "FAIL")
  console.log("Second enter blocked:", !secondEnter ? "PASS" : "FAIL")
  console.log("State:", mgr.getStatus())  // should be "in_position"
}

// === Test: Invalid State Transitions ===
function testInvalidTransitions() {
  console.log("\n=== Test: Invalid Transitions ===")

  const mgr = new PositionMachine("XYZ")

  // Should fail: cannot enter from idle
  const enterFromIdle = mgr.enter(100)
  console.log("Enter from idle blocked:", !enterFromIdle ? "PASS" : "FAIL")

  // Should fail: cannot exit from idle
  const exitFromIdle = mgr.exit()
  console.log("Exit from idle blocked:", !exitFromIdle ? "PASS" : "FAIL")

  // Now valid sequence
  mgr.watch()     // idle → watching ✓
  mgr.enter(100)  // watching → in_position ✓
  const exitOk = mgr.exit()  // in_position → exited ✓
  console.log("Exit from position succeeded:", exitOk ? "PASS" : "FAIL")
}

// === Test: State Preserves Through Pause/Resume ===
function testStatePersistence() {
  console.log("\n=== Test: State Persistence ===")

  const feed = new SimulatedFeed(["PERSIST"])
  const mgr = new PositionMachine("PERSIST")

  mgr.watch()
  mgr.enter(150)
  console.log("Before pause:", mgr.getStatus(), `(entry: ${mgr.entryPrice})`)

  feed.start()
  setTimeout(() => {
    feed.pause()
    console.log("During pause:", mgr.getStatus(), "(state survived disconnect)")

    feed.resume()
    setTimeout(() => {
      console.log("After resume:", mgr.getStatus(), "(state unchanged by reconnect)")
      feed.stop()
    }, 1000)
  }, 1500)
}

// === Test: Per-Symbol Routing ===
function testRouting() {
  console.log("\n=== Test: Per-Symbol Routing ===")

  const feed = new SimulatedFeed(["A", "B"])
  const mgrA = new PositionMachine("A")
  const mgrB = new PositionMachine("B")

  let aTickCount = 0
  let bTickCount = 0

  mgrA.watch()
  mgrB.watch()

  feed.on("tick", (tick: OHLC) => {
    if (tick.symbol === "A") aTickCount++
    if (tick.symbol === "B") bTickCount++
  })

  feed.start()

  setTimeout(() => {
    console.log(`Symbol A received ${aTickCount} ticks`)  // should be 2-3 (every 500ms for ~1s)
    console.log(`Symbol B received ${bTickCount} ticks`)  // should be same
    console.log("Routing:", aTickCount > 0 && bTickCount > 0 ? "PASS" : "FAIL")
    feed.stop()
  }, 1500)
}

// === Run All Tests ===
console.log("=== Lab C Test Suite ===\n")
testDoubleEntryPrevention()
testInvalidTransitions()
// testStatePersistence()  // async — run separately
// testRouting()           // async — run separately
```

## Done When

- [ ] All synchronous tests pass (double entry, invalid transitions)
- [ ] Async tests pass (state persistence, routing)
- [ ] Every "INVALID" log proves the state machine is working

## Common Mistake

```typescript
// ❌ Only testing happy path
test("enter works") {
  mgr.watch()
  mgr.enter(150)  // only tests success
}

// ✅ Test failure cases too
test("enter from idle fails") {
  const result = mgr.enter(150)
  expect(result).toBe(false)  // explicit failure test
}
```
