```typescript
// 6.0-final.ts
/**
 * Lab C — Final Assembly
 * Combines: Feed (3.3) + PositionMachine (2.3) + Routing (4.0) + Pause/Resume (5.0)
 *
 * Run: npx tsx 6.0-final.ts
 */

import { EventEmitter } from "events"

// ==================== TYPES ====================

interface OHLC {
  symbol: string
  open: number
  high: number
  low: number
  close: number
  timestamp: number
}

type Status = "idle" | "watching" | "in_position" | "exited"

interface SignalData {
  symbol: string
  direction: "long" | "short"
}

// ==================== STATE MACHINE ====================

const ALLOWED_TRANSITIONS: Record<Status, Status[]> = {
  idle: ["watching"],
  watching: ["in_position", "idle"],
  in_position: ["exited", "idle"],
  exited: ["idle"],
}

class PositionMachine {
  private status: Status = "idle"
  private entryPrice: number | null = null

  constructor(public symbol: string) {}

  getStatus(): Status {
    return this.status
  }

  private tryAction(action: string, price?: number): boolean {
    const actionToStatus: Record<string, Status> = {
      watch: "watching",
      enter: "in_position",
      exit: "exited",
      reset: "idle",
    }

    const target = actionToStatus[action]
    if (!target) {
      console.log(`  ❌ INVALID ACTION: ${action}`)
      return false
    }

    if (ALLOWED_TRANSITIONS[this.status].includes(target)) {
      console.log(`  ${this.symbol}: ${this.status} → ${target}`)
      this.status = target

      if (target === "in_position" && price !== undefined) {
        this.entryPrice = price
      }
      if (target === "idle") {
        this.entryPrice = null
      }
      return true
    }

    console.log(`  ❌ INVALID TRANSITION: ${this.symbol} cannot go ${this.status} → ${target}`)
    return false
  }

  onTick(price: number): void {
    console.log(`  [tick] ${this.symbol}: $${price.toFixed(2)} (${this.status})`)
  }

  onSignal(direction: "long" | "short"): void {
    if (this.status === "watching") {
      this.tryAction("enter")
    } else if (this.status === "in_position") {
      this.tryAction("exit")
    } else {
      console.log(`  ${this.symbol}: signal ignored (status: ${this.status})`)
    }
  }

  watch(): boolean { return this.tryAction("watch") }
  enter(price: number): boolean { return this.tryAction("enter", price) }
  exit(): boolean { return this.tryAction("exit") }
  reset(): boolean { return this.tryAction("reset") }
}

// ==================== FEED ====================

class SimulatedFeed extends EventEmitter {
  private prices: Map<string, number> = new Map()
  private ticker: NodeJS.Timeout | null = null
  private tickCount = 0
  private tickers: Map<string, boolean> = new Map()  // symbol → watching?

  constructor(private symbols: string[]) {
    super()
  }

  start(): void {
    if (this.ticker) return
    for (const s of this.symbols) {
      this.prices.set(s, 100)
      this.tickers.set(s, true)
    }

    this.ticker = setInterval(() => {
      this.tickCount++

      for (const symbol of this.symbols) {
        if (!this.tickers.get(symbol)) continue  // skip paused symbols

        const current = this.prices.get(symbol) ?? 100
        const change = (Math.random() - 0.5) * 2
        const next = Math.max(1, current + change)
        this.prices.set(symbol, next)

        this.emit("tick", {
          symbol,
          open: current,
          high: Math.max(current, next),
          low: Math.min(current, next),
          close: next,
          timestamp: Date.now(),
        })

        // Signal once every 5 ticks
        if (this.tickCount % 5 === 0 && Math.random() > 0.3) {
          this.emit("signal", {
            symbol,
            direction: Math.random() > 0.5 ? "long" : "short",
          })
        }
      }
    }, 500)

    console.log(`[feed] started: ${this.symbols.length} symbols at 500ms interval`)
  }

  pause(): void {
    if (this.ticker) {
      clearInterval(this.ticker)
      this.ticker = null
      this.emit("pause")
      console.log("[feed] paused (disconnect)")
    }
  }

  resume(): void {
    if (!this.ticker) {
      this.start()
      this.emit("resume")
      console.log("[feed] resumed (reconnect)")
    }
  }

  stop(): void {
    if (this.ticker) clearInterval(this.ticker)
    this.ticker = null
  }
}

// ==================== MAIN: WIRING ====================

const SYMBOLS = ["AAPL", "MSFT", "GOOGL"]

const feed = new SimulatedFeed(SYMBOLS)
const managers = new Map(SYMBOLS.map(s => [s, new PositionMachine(s)]))

// Symbol-based routing
feed.on("tick", (tick: OHLC) => {
  managers.get(tick.symbol)?.onTick(tick.close)
})

feed.on("signal", (signal: SignalData) => {
  managers.get(signal.symbol)?.onSignal(signal.direction)
})

feed.on("pause", () => {
  console.log(">>> Holding state during disconnect")
})

feed.on("resume", () => {
  console.log(">>> Receiving ticks after reconnect")
})

// Start
console.log("=== Lab C — Final Assembly ===\n")

for (const mgr of managers.values()) {
  mgr.watch()
}

feed.start()

// Demo sequence
setTimeout(() => {
  console.log("\n--- DEMO: Invalid transition (double watch) ---")
  const aapl = managers.get("AAPL")
  aapl?.watch()  // already watching → INVALID TRANSITION
}, 4000)

setTimeout(() => {
  console.log("\n--- DEMO: Pause (disconnect) ---")
  feed.pause()
  console.log(`AAPL state: ${managers.get("AAPL")?.getStatus()}`)
}, 6000)

setTimeout(() => {
  console.log("\n--- DEMO: Resume (reconnect) ---")
  feed.resume()
}, 8000)

// Cleanup
setTimeout(() => {
  console.log("\n=== FINAL STATES ===")
  for (const mgr of managers.values()) {
    console.log(`${mgr.symbol}: ${mgr.getStatus()}`)
  }
  feed.stop()
  process.exit(0)
}, 10000)
```
