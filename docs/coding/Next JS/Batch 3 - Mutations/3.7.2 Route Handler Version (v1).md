### `@/lib/reviews` Kya Hai?

Import mein `@/lib/reviews` likha hai. Ye **path alias** hai — `@/` project root ko refer karta hai.

```tsx
// Ye dono same hain
import { getAllReviews } from "@/lib/reviews";    // @/ alias se
import { getAllReviews } from "../../lib/reviews"; // relative path se
```

**Kyun use karte hain?** Kyunki `@/lib/reviews` padhne mein simple hai. Relative paths mein `../../` count karna padta hai — confusing ho jaata hai.

**Configuration:** `tsconfig.json` mein automatically set hota hai:
```json
{
  "paths": {
    "@/*": ["./*"]
  }
}
```

Ye mat samajho ki kuch karna hai — Next.js project mein ye pehle se set hota hai.

## Step 2: Route Handler — API Endpoints

```tsx
// app/api/reviews/route.ts
import { NextResponse } from "next/server";
import { getAllReviews, addReview } from "@/lib/reviews";

// GET /api/reviews — Saare reviews lao
export async function GET() {
  const reviews = getAllReviews();
  return NextResponse.json(reviews);
}

// POST /api/reviews — Naya review banao
export async function POST(request: Request) {
  const body = await request.json();

  // Validation
  if (!body.text || !body.productId || !body.rating) {
    return NextResponse.json(
      { error: "Text, productId, aur rating zaroori hai" },
      { status: 400 }
    );
  }

  const review = addReview({
    productId: body.productId,
    text: body.text,
    rating: body.rating,
  });

  return NextResponse.json(review, { status: 201 });
}
```

```tsx
// app/api/reviews/[id]/route.ts
import { NextResponse } from "next/server";
import { deleteReview } from "@/lib/reviews";

// DELETE /api/reviews/1 — Review delete karo
export async function DELETE(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  const deleted = deleteReview(Number(id));

  if (!deleted) {
    return NextResponse.json({ error: "Review nahi mila" }, { status: 404 });
  }

  return NextResponse.json({ message: "Review delete ho gaya" });
}
```

## Step 3: Server Component — Review List

```tsx
// app/components/ReviewList.tsx
import { getReviewsByProduct } from "@/lib/reviews";

export function ReviewList({ productId }: { productId: number }) {
  const reviews = getReviewsByProduct(productId);

  if (reviews.length === 0) {
    return <p className="text-gray-500">Abhi koi review nahi hai.</p>;
  }

  return (
    <div className="space-y-2">
      <h3 className="font-bold">Reviews ({reviews.length})</h3>
      {reviews.map((review) => (
        <div key={review.id} className="border p-2 rounded">
          <div className="flex justify-between">
            <span>{"⭐".repeat(review.rating)}</span>
            <span className="text-sm text-gray-500">{review.createdAt}</span>
          </div>
          <p>{review.text}</p>
        </div>
      ))}
    </div>
  );
}
```

## Step 4: Client Component — Review Form (Manual Pending State)

```tsx
// app/components/ReviewForm.tsx
"use client";

import { useState } from "react";

export function ReviewForm({ productId }: { productId: number }) {
  const [text, setText] = useState("");
  const [rating, setRating] = useState(5);
  const [pending, setPending] = useState(false);
  const [message, setMessage] = useState("");

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setPending(true);
    setMessage("");

    try {
      const res = await fetch("/api/reviews", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ productId, text, rating }),
      });

      if (!res.ok) {
        throw new Error("Failed to add review");
      }

      const review = await res.json();
      setMessage("Review add ho gaya!");
      setText("");
      setRating(5);

      // Page refresh karo (manual)
      // window.location.reload() — poori page reload hoti hai
      // Problem: User ka scroll position ud jaata hai, form ka data ud jaata hai
      // Better: Revalidation (3.6 mein seekha) — sirf data refresh hota hai
      window.location.reload();
    } catch (error) {
      setMessage("Error: Review add nahi ho paya");
    } finally {
      setPending(false);
    }
  };

  return (
    <form onSubmit={handleSubmit} className="space-y-2 border p-4 rounded">
      <h3 className="font-bold">Naya Review Likho</h3>

      <div>
        <label>Rating:</label>
        <select
          value={rating}
          onChange={(e) => setRating(Number(e.target.value))}
          className="border p-1 rounded ml-2"
        >
          {[1, 2, 3, 4, 5].map((n) => (
            <option key={n} value={n}>
              {"⭐".repeat(n)}
            </option>
          ))}
        </select>
      </div>

      <textarea
        value={text}
        onChange={(e) => setText(e.target.value)}
        placeholder="Apna review likho..."
        className="w-full border p-2 rounded"
        rows={3}
      />

      <button
        type="submit"
        disabled={pending}
        className="bg-blue-500 text-white px-4 py-2 rounded disabled:bg-gray-400"
      >
        {pending ? "Loading..." : "Submit Review"}
      </button>

      {message && (
        <p className={message.startsWith("Error") ? "text-red-500" : "text-green-500"}>
          {message}
        </p>
      )}
    </form>
  );
}
```

## Route Handler Version Ka Summary

| Step | Kya Kiya |
|------|----------|
| 1 | Data layer banaya (reviews.ts) |
| 2 | Route Handler banaya (GET, POST, DELETE) |
| 3 | Server Component — review list dikhata hai |
| 4 | Client Component — form + manual pending state |
| 5 | Main page — sab jod diya |

**Boilerplate dekho:**
- Route file (2 files — `[id]/route.ts` aur `route.ts`)
- Client mein `fetch()`, headers, JSON.stringify
- Manual `useState` for pending state
- Manual `window.location.reload()` for refresh
