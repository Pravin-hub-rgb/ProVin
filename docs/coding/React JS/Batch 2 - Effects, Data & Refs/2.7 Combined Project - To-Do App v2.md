## Step 3 — ThemeContext (Context API)

```tsx
// context/ThemeContext.tsx
import { createContext, useContext, useState, useCallback, useMemo, type ReactNode } from "react";

type Theme = "light" | "dark";

interface ThemeContextValue {
  theme: Theme;
  toggleTheme: () => void;
}

const ThemeContext = createContext<ThemeContextValue | undefined>(undefined);

export function useTheme(): ThemeContextValue {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error("useTheme must be used within ThemeProvider");
  return ctx;
}

export function ThemeProvider({ children }: { children: ReactNode }) {
  const [theme, setTheme] = useState<Theme>("light");
  const toggleTheme = useCallback(() => {
    setTheme((prev) => (prev === "light" ? "dark" : "light"));
  }, []);

  const value = useMemo(() => ({ theme, toggleTheme }), [theme]);

  return <ThemeContext value={value}>{children}</ThemeContext>;
}
```

```tsx
// components/TaskInput.tsx
import { useState, useRef, useEffect } from "react";

interface TaskInputProps {
  onAdd: (text: string) => void;
}

function TaskInput({ onAdd }: TaskInputProps): JSX.Element {
  const [text, setText] = useState("");
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    inputRef.current?.focus();
  }, []);

  const handleClick = () => {
    if (text.trim() !== "") {
      onAdd(text.trim());
      setText("");
      inputRef.current?.focus();
    }
  };

  return (
    <div>
      <input
        ref={inputRef}
        type="text"
        value={text}
        onChange={(e) => setText(e.target.value)}
        onKeyDown={(e) => e.key === "Enter" && handleClick()}
        placeholder="Add a task..."
      />
      <button onClick={handleClick}>Add</button>
    </div>
  );
}

export default TaskInput;
```

## Step 5 — App.tsx (All Concepts Together)

```tsx
import { useState, useEffect } from "react";
import { ThemeProvider, useTheme } from "./context/ThemeContext";
import TaskInput from "./components/TaskInput";
import TaskItem from "./components/TaskItem";
import FilterBar from "./components/FilterBar";
import TaskStats from "./components/TaskStats";
import ThemeToggle from "./components/ThemeToggle";
import type { Task, FilterStatus } from "./types";

// Categories API
async function fetchCategories(): Promise<string[]> {
  const res = await fetch(
    "https://www.themealdb.com/api/json/v1/1/categories.php"
  );
  const data = await res.json();
  return data.categories.map((c: any) => c.strCategory).slice(0, 5);
}

function ToDoApp() {
  const [tasks, setTasks] = useState<Task[]>([
    { id: 1, text: "Learn useEffect", completed: true, category: "Learning" },
    { id: 2, text: "Build Recipe Finder", completed: true, category: "Projects" },
    { id: 3, text: "Master useRef", completed: false, category: "Learning" },
    { id: 4, text: "Deploy To-Do App v2", completed: false, category: "Projects" },
  ]);
  const [filter, setFilter] = useState<FilterStatus>("all");
  const [categories, setCategories] = useState<string[]>([]);
  const { theme } = useTheme();

  // useEffect: fetch categories on mount
  useEffect(() => {
    let ignore = false;
    fetchCategories().then((cats) => {
      if (!ignore) setCategories(cats);
    });
    return () => { ignore = true; };
  }, []);

  const handleAdd = (text: string) => {
    const category = categories[Math.floor(Math.random() * categories.length)] || "General";
    setTasks([...tasks, { id: Date.now(), text, completed: false, category }]);
  };

  const handleToggle = (id: number) => {
    setTasks(tasks.map((t) =>
      t.id === id ? { ...t, completed: !t.completed } : t
    ));
  };

  const handleDelete = (id: number) => {
    setTasks(tasks.filter((t) => t.id !== id));
  };

  const filteredTasks = tasks.filter((t) => {
    if (filter === "active") return !t.completed;
    if (filter === "completed") return t.completed;
    return true;
  });

  const completedCount = tasks.filter((t) => t.completed).length;
  const remainingCount = tasks.length - completedCount;

  const styles: Record<string, React.CSSProperties> = {
    app: {
      padding: "20px",
      maxWidth: "600px",
      margin: "0 auto",
      background: theme === "dark" ? "#1a1a2e" : "#fff",
      color: theme === "dark" ? "#fff" : "#000",
      minHeight: "100vh",
      transition: "all 0.3s",
    },
  };

  return (
    <div style={styles.app}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
        <h1>📋 To-Do App v2</h1>
        <ThemeToggle />
      </div>

      {categories.length > 0 && (
        <p style={{ fontSize: "13px", color: "#888" }}>
          Categories loaded: {categories.join(", ")}
        </p>
      )}

      <TaskInput onAdd={handleAdd} />
      <FilterBar currentFilter={filter} onFilterChange={setFilter} />

      <div style={{ marginTop: "16px" }}>
        {filteredTasks.map((task) => (
          <TaskItem
            key={task.id}
            task={task}
            onToggle={handleToggle}
            onDelete={handleDelete}
          />
        ))}
        {filteredTasks.length === 0 && <p>No tasks to show.</p>}
      </div>

      <TaskStats
        total={tasks.length}
        completed={completedCount}
        remaining={remainingCount}
      />
    </div>
  );
}

// Wrap with ThemeProvider
function App() {
  return (
    <ThemeProvider>
      <ToDoApp />
    </ThemeProvider>
  );
}

export default App;
```

## Khud Try Karo

1. Code likho — ThemeContext mein ek `isDark` helper add karo jo boolean return kare, aur use karo styling mein.
   
   <details>
   <summary>Sample Answer</summary>
   Context value mein `isDark: theme === "dark"` add karo. Phir `const { isDark } = useTheme();` se kahi bhi use karo — `background: isDark ? "#1a1a2e" : "#fff"`.
   </details>

2. Predict karo — agar `TaskInput` ka `useEffect([])` hata do toh kya hoga?

   <details>
   <summary>Sample Answer</summary>
   Input pe auto-focus nahi hoga. User ko manually input click karna padega type karne se pehle. UseEffect har render ke baad focus karta hai — lekin `[]` hone ki wajah se sirf mount pe ek baar focus hota hai.
   </details>

3. Modify karo — `categories` fetch mein error state bhi add karo (abhi sirf `setCategories` hai, error handle nahi hai).

   <details>
   <summary>Sample Answer</summary>
   `const [catError, setCatError] = useState<string | null>(null)` add karo. Fetch mein `try/catch` daalo, error pe `setCatError(err.message)`. UI mein `{catError && <p>❌ {catError}</p>}` dikhao.
   </details>
