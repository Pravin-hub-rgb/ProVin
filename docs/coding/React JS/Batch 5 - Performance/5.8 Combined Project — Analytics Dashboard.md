```tsx
// types.ts
export interface Sale {
  id: number
  product: string
  category: string
  amount: number
  date: string
  region: string
}

export type SortField = 'product' | 'amount' | 'date' | 'region'
export type SortOrder = 'asc' | 'desc'

export interface FilterState {
  search: string
  category: string
  region: string
  sortField: SortField
  sortOrder: SortOrder
  page: number
}

export type FilterAction =
  | { type: 'SET_SEARCH'; search: string }
  | { type: 'SET_CATEGORY'; category: string }
  | { type: 'SET_REGION'; region: string }
  | { type: 'SET_SORT'; field: SortField }
  | { type: 'TOGGLE_ORDER' }
  | { type: 'SET_PAGE'; page: number }
  | { type: 'RESET_FILTERS' }
```

## Step 3 — useReducer for Filter State

```tsx
// useFilterReducer.ts
import { useReducer } from 'react'
import type { FilterState, FilterAction } from './types'

const initialState: FilterState = {
  search: '',
  category: 'all',
  region: 'all',
  sortField: 'date',
  sortOrder: 'desc',
  page: 1,
}

function filterReducer(state: FilterState, action: FilterAction): FilterState {
  switch (action.type) {
    case 'SET_SEARCH':
      return { ...state, search: action.search, page: 1 }
    case 'SET_CATEGORY':
      return { ...state, category: action.category, page: 1 }
    case 'SET_REGION':
      return { ...state, region: action.region, page: 1 }
    case 'SET_SORT':
      return {
        ...state,
        sortField: action.field,
        sortOrder: state.sortField === action.field
          ? (state.sortOrder === 'asc' ? 'desc' : 'asc')
          : 'asc',
        page: 1,
      }
    case 'TOGGLE_ORDER':
      return { ...state, sortOrder: state.sortOrder === 'asc' ? 'desc' : 'asc' }
    case 'SET_PAGE':
      return { ...state, page: action.page }
    case 'RESET_FILTERS':
      return initialState
    default:
      return state
  }
}

export function useFilterReducer() {
  return useReducer(filterReducer, initialState)
}
```

Notice: `SET_SEARCH`, `SET_CATEGORY`, `SET_REGION` — these change a filter and reset the page to 1. `SET_SORT` — toggles the sort order if it's the same field. All logic in one place.

## Step 4 — useMemo for Derived Data

```tsx
// useSalesData.ts
import { useMemo } from 'react'
import type { Sale, FilterState } from './types'

const PAGE_SIZE = 10

export function useSalesData(
  sales: Sale[],
  filters: FilterState
) {
  // Filter + sort + paginate — all memed together
  const processedData = useMemo(() => {
    let result = [...sales]

    // Search filter
    if (filters.search) {
      const q = filters.search.toLowerCase()
      result = result.filter(s =>
        s.product.toLowerCase().includes(q) ||
        s.region.toLowerCase().includes(q)
      )
    }

    // Category filter
    if (filters.category !== 'all') {
      result = result.filter(s => s.category === filters.category)
    }

    // Region filter
    if (filters.region !== 'all') {
      result = result.filter(s => s.region === filters.region)
    }

    // Sort
    result.sort((a, b) => {
      const dir = filters.sortOrder === 'asc' ? 1 : -1
      switch (filters.sortField) {
        case 'product': return a.product.localeCompare(b.product) * dir
        case 'amount': return (a.amount - b.amount) * dir
        case 'date': return a.date.localeCompare(b.date) * dir
        case 'region': return a.region.localeCompare(b.region) * dir
        default: return 0
      }
    })

    return result
  }, [sales, filters])

  // Derived metrics — cached
  const metrics = useMemo(() => ({
    totalSales: processedData.reduce((sum, s) => sum + s.amount, 0),
    averageSale: processedData.length
      ? Math.round(processedData.reduce((sum, s) => sum + s.amount, 0) / processedData.length)
      : 0,
    totalItems: processedData.length,
  }), [processedData])

  // Pagination
  const totalPages = Math.ceil(processedData.length / PAGE_SIZE)
  const paginatedData = useMemo(
    () => processedData.slice(
      (filters.page - 1) * PAGE_SIZE,
      filters.page * PAGE_SIZE
    ),
    [processedData, filters.page]
  )

  return { paginatedData, metrics, totalPages }
}
```

Every derived value is cached with `useMemo`. Data is only processed when `sales` or `filters` change. All other re-renders use the cached value.

## Step 5 — React.memo for Table Rows

```tsx
// SalesRow.tsx
import { memo } from 'react'
import type { Sale } from './types'

interface SalesRowProps {
  sale: Sale
  onRowClick: (id: number) => void
}

const SalesRow = memo(function SalesRow({ sale, onRowClick }: SalesRowProps) {
  return (
    <tr
      onClick={() => onRowClick(sale.id)}
      className="border-b hover:bg-gray-50 cursor-pointer"
    >
      <td className="p-3">{sale.product}</td>
      <td className="p-3">{sale.category}</td>
      <td className="p-3 text-right">${sale.amount.toFixed(2)}</td>
      <td className="p-3">{sale.date}</td>
      <td className="p-3">{sale.region}</td>
    </tr>
  )
})

export default SalesRow
```

## Step 6 — useCallback for Row Click Handler

```tsx
// Dashboard.tsx
import { useCallback } from 'react'

function Dashboard() {
  const [filters, dispatch] = useFilterReducer()

  const handleRowClick = useCallback((id: number) => {
    console.log('Row clicked:', id)
    // Navigate to detail page, or open a modal
  }, [])
  // Empty dependencies — the function will never change

  return (
    <table>
      <tbody>
        {paginatedData.map(sale => (
          <SalesRow
            key={sale.id}
            sale={sale}
            onRowClick={handleRowClick}
          />
        ))}
      </tbody>
    </table>
  )
}
```

`handleRowClick`'s reference stays stable (empty deps). `SalesRow` is memoized — as long as the `sale` object doesn't change (which only happens when the actual data changes), the row won't re-render.

## Step 7 — React.lazy for Chart Component

Lazy-load the chart component — chart libraries tend to be heavy:

```tsx
// ChartSection.tsx — this will be lazily loaded
import { useMemo } from 'react'
import type { Sale } from './types'

interface ChartSectionProps {
  data: Sale[]
}

export default function ChartSection({ data }: ChartSectionProps) {
  const chartData = useMemo(() => {
    // Process data for charts
    const byCategory: Record<string, number> = {}
    data.forEach(s => {
      byCategory[s.category] = (byCategory[s.category] || 0) + s.amount
    })
    return Object.entries(byCategory).map(([category, total]) => ({
      category,
      total,
    }))
  }, [data])

  return (
    <div className="bg-white p-6 rounded-lg shadow">
      <h3 className="text-lg font-bold mb-4">Sales by Category</h3>
      {/* Simple bar chart with divs */}
      <div className="space-y-2">
        {chartData.map(d => (
          <div key={d.category} className="flex items-center gap-4">
            <span className="w-24 text-sm">{d.category}</span>
            <div className="flex-1 bg-gray-200 rounded h-6">
              <div
                className="bg-blue-500 rounded h-6 transition-all"
                style={{ width: `${(d.total / Math.max(...chartData.map(x => x.total))) * 100}%` }}
              />
            </div>
            <span className="text-sm font-mono w-20 text-right">${d.total}</span>
          </div>
        ))}
      </div>
    </div>
  )
}
```

Lazy import in Dashboard:

```tsx
import { lazy, Suspense } from 'react'
const ChartSection = lazy(() => import('./ChartSection'))

// When the user clicks "Show Charts":
{showCharts && (
  <Suspense fallback={<div className="p-8 text-center">Charts loading...</div>}>
    <ChartSection data={processedData} />
  </Suspense>
)}
```

The ChartSection code won't be in the initial bundle — it only downloads when the user clicks "Show Charts".

## Step 8 — The Full Dashboard Component

```tsx
// Dashboard.tsx
import { useState, useCallback, Suspense, lazy } from 'react'
import SalesRow from './SalesRow'
import { useSalesData } from './useSalesData'
import { useFilterReducer } from './useFilterReducer'
import { salesData } from './mockData'

const ChartSection = lazy(() => import('./ChartSection'))

export default function Dashboard() {
  const [showCharts, setShowCharts] = useState(false)
  const [filters, dispatch] = useFilterReducer()
  const { paginatedData, metrics, totalPages } = useSalesData(salesData, filters)

  const handleRowClick = useCallback((id: number) => {
    console.log('Row:', id)
  }, [])

  return (
    <div className="p-8 max-w-6xl mx-auto">
      {/* Metrics Cards */}
      <div className="grid grid-cols-3 gap-4 mb-8">
        <div className="bg-white p-4 rounded-lg shadow">
          <p className="text-sm text-gray-500">Total Sales</p>
          <p className="text-2xl font-bold">${metrics.totalSales.toFixed(2)}</p>
        </div>
        <div className="bg-white p-4 rounded-lg shadow">
          <p className="text-sm text-gray-500">Average Sale</p>
          <p className="text-2xl font-bold">${metrics.averageSale.toFixed(2)}</p>
        </div>
        <div className="bg-white p-4 rounded-lg shadow">
          <p className="text-sm text-gray-500">Items Found</p>
          <p className="text-2xl font-bold">{metrics.totalItems}</p>
        </div>
      </div>

      {/* Filters */}
      <div className="flex gap-4 mb-6">
        <input
          className="border rounded px-3 py-2"
          placeholder="Search products..."
          value={filters.search}
          onChange={e => dispatch({ type: 'SET_SEARCH', search: e.target.value })}
        />
        <select
          className="border rounded px-3 py-2"
          value={filters.category}
          onChange={e => dispatch({ type: 'SET_CATEGORY', category: e.target.value })}
        >
          <option value="all">All Categories</option>
          <option value="electronics">Electronics</option>
          <option value="clothing">Clothing</option>
          <option value="food">Food</option>
        </select>
        <select
          className="border rounded px-3 py-2"
          value={filters.region}
          onChange={e => dispatch({ type: 'SET_REGION', region: e.target.value })}
        >
          <option value="all">All Regions</option>
          <option value="north">North</option>
          <option value="south">South</option>
          <option value="east">East</option>
          <option value="west">West</option>
        </select>
      </div>

      {/* Table */}
      <div className="bg-white rounded-lg shadow overflow-hidden mb-6">
        <table className="w-full">
          <thead className="bg-gray-50">
            <tr>
              <th className="p-3 text-left cursor-pointer" onClick={() => dispatch({ type: 'SET_SORT', field: 'product' })}>
                Product {filters.sortField === 'product' && (filters.sortOrder === 'asc' ? '↑' : '↓')}
              </th>
              <th className="p-3 text-left">Category</th>
              <th className="p-3 text-right cursor-pointer" onClick={() => dispatch({ type: 'SET_SORT', field: 'amount' })}>
                Amount {filters.sortField === 'amount' && (filters.sortOrder === 'asc' ? '↑' : '↓')}
              </th>
              <th className="p-3 text-left cursor-pointer" onClick={() => dispatch({ type: 'SET_SORT', field: 'date' })}>
                Date {filters.sortField === 'date' && (filters.sortOrder === 'asc' ? '↑' : '↓')}
              </th>
              <th className="p-3 text-left">Region</th>
            </tr>
          </thead>
          <tbody>
            {paginatedData.map(sale => (
              <SalesRow key={sale.id} sale={sale} onRowClick={handleRowClick} />
            ))}
          </tbody>
        </table>
      </div>

      {/* Pagination */}
      <div className="flex justify-center gap-2 mb-8">
        {Array.from({ length: totalPages }, (_, i) => (
          <button
            key={i}
            onClick={() => dispatch({ type: 'SET_PAGE', page: i + 1 })}
            className={`px-3 py-1 rounded ${filters.page === i + 1 ? 'bg-blue-500 text-white' : 'bg-gray-200'}`}
          >
            {i + 1}
          </button>
        ))}
      </div>

      {/* Lazy Charts */}
      <button
        onClick={() => setShowCharts(s => !s)}
        className="bg-green-500 text-white px-6 py-2 rounded mb-6"
      >
        {showCharts ? 'Hide Charts' : 'Show Charts'}
      </button>

      {showCharts && (
        <Suspense fallback={<div className="p-8 text-center">Charts loading...</div>}>
          <ChartSection data={salesData} />
        </Suspense>
      )}
    </div>
  )
}
```

## Summary

✅ **useReducer** — all filter/sort/pagination state in one central reducer  
✅ **useMemo** — derived data (filtered, sorted, paginated) and metrics are cached  
✅ **React.memo** — `SalesRow` rows don't re-render until their data actually changes  
✅ **useCallback** — the row click handler has a stable reference, so memo works  
✅ **React.lazy + Suspense** — `ChartSection` only loads when the user clicks "Show Charts"  
✅ **Profiler** — verified with the flamegraph that the optimization is working
